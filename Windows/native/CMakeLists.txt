cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(HlgBridgeCodec LANGUAGES C CXX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(UHDR_BUILD_DEPS ON CACHE BOOL "" FORCE)
set(UHDR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(UHDR_ENABLE_HEIF OFF CACHE BOOL "" FORCE)
set(UHDR_WRITE_XMP ON CACHE BOOL "" FORCE)
set(UHDR_WRITE_ISO ON CACHE BOOL "" FORCE)
FetchContent_Declare(ultrahdr
    GIT_REPOSITORY https://github.com/google/libultrahdr.git
    GIT_TAG e5f5a022fe96fc4dc2ee35c19f733a50df807abe)
FetchContent_MakeAvailable(ultrahdr)
add_library(hlgcodec SHARED ../../Sources/CHDR/CHDR.c)
target_include_directories(hlgcodec PRIVATE ../../Sources/CHDR/include "${ultrahdr_SOURCE_DIR}")
target_compile_definitions(hlgcodec PRIVATE HLGBRIDGE_EXPORTS)
if(TARGET uhdr-static)
    target_link_libraries(hlgcodec PRIVATE uhdr-static)
else()
    target_link_libraries(hlgcodec PRIVATE uhdr)
endif()
if(MSVC)
    target_compile_options(hlgcodec PRIVATE /utf-8)
endif()
